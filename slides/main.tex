\section{Basic Task A: Unconditional Generation}

\begin{frame}[t]{Basic A (1): token representation --- REMI}
  \begin{tight}
    \item REMI (Huang \& Yang, 2020): \texttt{Bar}, \texttt{Pos} (16th note), \texttt{Chord}, \texttt{Tempo}, \texttt{Pitch}, \texttt{Vel}, \texttt{Dur}
    \item Bars follow the downbeats annotated in \texttt{beat\_midi.txt}; a \texttt{Beat\_n} token gives each bar's length
    \item MELODY, BRIDGE and PIANO merged into one stream; 430 tokens in the vocabulary
    \item 32 bars $\approx$ 2{,}431 tokens (median)
  \end{tight}
  \vspace{1em}
  \textbf{Example} (song 001, bar 3)\\[0.3em]
  {\footnotesize\ttfamily Bar Beat\_4 Tempo\_90 Pos\_0 Chord\_A\#:min Pitch\_46 Vel\_21 Dur\_7
  Pitch\_75 Vel\_29 Dur\_2 Pos\_1 Pitch\_53 Vel\_18 Dur\_5 Pos\_2 \dots\par}
\end{frame}

\begin{frame}[t]{Basic A (2): model architecture}
  \centering
  \resizebox{\linewidth}{!}{%
  \begin{tikzpicture}[node distance=0.55cm]
    \node[io, text width=2.0cm] (in) {\textbf{POP909 song}\\MIDI + beats\\+ chords};
    \node[op, text width=2.2cm, right=of in] (tok) {\textbf{REMI tokens}\\random 32-bar window\\$\pm$5 semitones};

    \draw[arr] (in) -- (tok); \draw[arr] (tok) -- (gpt); \draw[arr] (gpt) -- (smp); \draw[arr] (smp) -- (out);

  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \small
      \begin{tabular}{@{}ll@{}}

\end{frame}

        Setting & Sampling & Temperature \\ \midrule
        A & top-k, $K = 20$ & 0.9 \\
        B & top-p, $P = 0.9$ & 1.0 \\
        C & top-k, $K = 50$ & 1.2 \\
        \bottomrule
      \end{tabular}\\[0.6em]
      \begin{tight}
        \item Checkpoint: \texttt{models/basic.pt} (step 5{,}000)
        \item 20 pieces $\times$ 32 bars per setting, piece $i$ uses seed $2026 + i$
        \item Ablation: the model trained \textbf{without chord tokens}, sampled with setting A
      \end{tight}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Grammar mask}
      \begin{tight}\small
        \item Sampling is restricted to tokens that can come next in REMI, e.g.\ \texttt{Pitch} $\rightarrow$ \texttt{Vel} $\rightarrow$ \texttt{Dur}
        \item \texttt{Pos} strictly increases inside a bar and stays below its beat count
        \item Generation stops after exactly 32 bars
      \end{tight}
    \end{column}
  \end{columns}
\end{frame}

\section{Bonus: Accompaniment Generation}
\begin{frame}[t]{Bonus (1): model and representation --- Option B, seq2seq}
  \centering
  \resizebox{\linewidth}{!}{%
  \begin{tikzpicture}[node distance=0.55cm]
    \node[io, text width=2.0cm] (in) {\textbf{melody.mid}\\+ beat\_midi.txt};
    \node[op, text width=2.6cm, right=of in] (src) {\textbf{MELODY tokens}\\\texttt{Bar Beat Tempo}\\\texttt{Pos Pitch Vel Dur}};
    \node[train, text width=2.4cm, right=of src] (enc) {\textbf{T5 encoder}\\6 layers, $d{=}512$\\sees all 16 bars};
    \node[train, text width=2.4cm, right=of enc] (dec) {\textbf{T5 decoder}\\6 layers, $d{=}512$\\cross-attention};
    \node[op, text width=2.4cm, right=of dec] (tgt) {\textbf{PIANO tokens}\\\texttt{Bar Pos Pitch Vel Dur}\\grammar mask};
    \node[io, text width=2.0cm, right=of tgt] (out) {\textbf{MELODY + PIANO}\\MIDI $\rightarrow$ WAV};
    \draw[arr] (in) -- (src); \draw[arr] (src) -- (enc); \draw[arr] (enc) -- (dec); \draw[arr] (dec) -- (tgt); \draw[arr] (tgt) -- (out);
  \end{tikzpicture}}

  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \footnotesize
      \begin{tabular}{@{}ll@{}}
        \toprule
        Model & \texttt{T5ForConditionalGeneration} \\
        Layers & 6 encoder + 6 decoder \\
        Width / heads / FFN & 512 / 8 / 2{,}048 \\
        Max length & 1{,}024 melody / 2{,}048 piano \\
        Parameters & \textbf{44.3 M}, random init \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.5\textwidth}
      \footnotesize
      \begin{tight}
        \item Same REMI vocabulary as Basic, but no chord tokens: the harmony must come from the melody alone
        \item Encoder: MELODY track with the bar headers, so it knows meter and tempo
        \item Decoder: PIANO track; bar headers are omitted because the piano is placed on the prompt's own beat grid
        \item Inference reads only \texttt{melody.mid} and \texttt{beat\_midi.txt}; the melody is kept verbatim
        \item 16 bars: median 316 melody / 758 piano tokens
      \end{tight}
    \end{column}
  \end{columns}
\end{frame}
